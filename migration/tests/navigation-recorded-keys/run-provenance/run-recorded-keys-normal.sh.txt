#!/bin/bash
set -euo pipefail
D=/workspace/shared/gimp-native-secondary
R=/workspace/scratch/0015fc5d1a03/gimp-painter
source "$D/env.sh";exec 9>"$D/build.lock";flock 9
python3 - <<'PY'
import pathlib,json,hashlib
D=pathlib.Path('/workspace/shared/gimp-native-secondary');R=pathlib.Path('/workspace/scratch/0015fc5d1a03/gimp-painter')
for row in json.loads((R/'migration/tests/navigation-item-geometry/held-source-isolation.json').read_text())['rows']:
 p=D/'source'/row['path']
 if row.get('native_exists') is False:assert not p.exists()
 else:assert hashlib.sha256(p.read_bytes()).hexdigest()==row['sha256'],row['path']
for path in ['app/tests/test-painter-navigation-events.c','app/tests/test-painter-navigation-recorded-keys.inc']:
 (D/'source'/path).write_bytes((R/path).read_bytes())
paths=list(json.loads((R/'migration/tests/navigation-reset-action/manifest.json').read_text())['source_sha256'])
paths+=['app/tests/test-painter-navigation-recorded-keys.inc','app/core/gimpprojection.c','migration/fixtures/legacy-rotation-operations/gui-captures/rotation.tsv','migration/fixtures/legacy-zoom/zoom.tsv']
for path in paths:assert (R/path).read_bytes()==(D/'source'/path).read_bytes(),path
out={p:hashlib.sha256((D/'source'/p).read_bytes()).hexdigest() for p in paths}
(R/'migration/tests/navigation-recorded-keys/source-before-build.json').write_text(json.dumps(out,indent=2)+'\n')
PY
ninja -j2 -C "$D/build" app/tests/painter-navigation-events > "$D/recorded-keys-build.log" 2>&1
set +e
GIMP_TEST_NAVIGATION_DRAW_DIR="$D/pan-screenshots" python3 "$D/source/tools/run-painter-xvfb.py" meson test -C "$D/build" --no-rebuild --print-errorlogs --test-args='-p /navigation-events/recorded-keys' painter-navigation-events > "$D/recorded-keys-normal.log" 2>&1
status=$?;set -e
python3 "$D/record-keys.py" build normal.json
exit "$status"
